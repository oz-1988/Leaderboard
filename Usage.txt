How to use in any project (Phaser or Vanilla)
1) Initialize once
import { Scoreboard } from "./scoreboard/Scoreboard.js";

export const scoreboard = new Scoreboard({
  apiBase: "https://your-api.com",
  gameSlug: "stacker",
  leaderboardLimit: 10,
  ui: {
    cssUrl: "./scoreboard/scoreboard.css",
    modalTemplateUrl: "./scoreboard/scoreboard.modal.html",
    leaderboardTemplateUrl: "./scoreboard/scoreboard.leaderboard.html",
  },
});

2) Call on end
await scoreboard.onGameEnd({ score: finalScore });
